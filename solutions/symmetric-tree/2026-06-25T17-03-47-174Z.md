class Solution {
    public boolean isSymmetric(TreeNode root) {
        return isMirror(root, root);
    }

    public boolean isMirror(TreeNode t1, TreeNode t2) {
        boolean result = false;
        if (t1 == null && t2 == null) {
            return true;
        } 

        if (t1 == null || t2 ==null) {
            return false;
        }

        if (t1.val != t2.val) {
            return false;
        }

        return isMirror(t1.right, t2.left) && isMirror(t1.left, t2.right);
    }
}
