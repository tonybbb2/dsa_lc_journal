class Solution {
    public List<Integer> inorderTraversal(TreeNode root) {
        List<Integer> result = new ArrayList<Integer>();

        inOrder(root, result);

        return result;
    }

    public void inOrder(TreeNode l1, List<Integer> resultt) {
        if (l1 == null) {
            return;
        }

        inOrder(l1.left, resultt);
        resultt.add(l1.val);
        inOrder(l1.right, resultt);
    }
}
